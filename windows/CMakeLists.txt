# The Windows app with MinGW, MSVC or clang-retro (the Windows 95 build:
# cmake/clang-retro.cmake).  ANSI, for the same code to build with all.
enable_language(CXX)
enable_language(RC)
include(sources.cmake)
# (the target dmodem-app, its file dmodem.exe: D-Modem's libdmodem is
# the target dmodem, with the modem built in)
add_executable(dmodem-app WIN32 ${DMODEM_WIN_SOURCES} res/app.rc)
set_target_properties(dmodem-app PROPERTIES OUTPUT_NAME dmodem)
target_include_directories(dmodem-app PRIVATE src)
target_link_libraries(dmodem-app PRIVATE dmodemctl comctl32 wsock32 comdlg32 gdi32 winmm msacm32 shell32 advapi32)
# the Remote page's SSH: libdmodemssh (libssh, built in)
if(TARGET dmodemssh)
  target_link_libraries(dmodem-app PRIVATE dmodemssh)
  target_compile_definitions(dmodem-app PRIVATE DMODEM_HAVE_SSH)
endif()
if(MSVC)
  target_compile_definitions(dmodem-app PRIVATE _CRT_SECURE_NO_WARNINGS)
  # (a manifest of our own in app.rc)
  target_link_options(dmodem-app PRIVATE /MANIFEST:NO)
else()
  target_compile_options(dmodem-app PRIVATE -Wall -Wextra -Wno-unused-parameter)
  target_link_options(dmodem-app PRIVATE -static)
endif()
# (picolibc starts main, unless the entry point is WinMain's)
if(CLANG_RETRO_CONFIG MATCHES "picolibc")
  target_link_options(dmodem-app PRIVATE -Wl,-e,WinMainCRTStartup)
endif()
# the modem built in (libdmodem_embed, DMODEM_GUI_EMBED): Windows 98 SE
# and later
if(DMODEM_GUI_EMBED AND TARGET dmodem_embed)
  target_link_libraries(dmodem-app PRIVATE dmodem_embed)
  target_compile_definitions(dmodem-app PRIVATE DMODEM_GUI_EMBED)
endif()
